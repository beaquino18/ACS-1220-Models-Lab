Book.query.all()

## Filter all books with audience for YOUNG_ADULT
Book.query.filter_by(audience=Audience.YOUNG_ADULT).all()

## Adding favorite books to current username - hermioneG
with app.app_context():
    # Get the specific user by username
    user = User.query.filter_by(username='hermioneG').first()
    
    if not user:
        print("User not found!")
    else:
        # Get the books you want to add
        book1 = Book.query.filter_by(title="1984").first()
        book2 = Book.query.filter_by(title="The Handmaid's Tale").first()
        
        if not book1 or not book2:
            print("One or both books not found!")
        else:
            # Add the books to the user's favorites if not already there
            if book1 not in user.favorite_books:
                user.favorite_books.append(book1)
                print(f"Added '{book1.title}' to {user.username}'s favorites")
            else:
                print(f"'{book1.title}' is already in {user.username}'s favorites")
                
            if book2 not in user.favorite_books:
                user.favorite_books.append(book2)
                print(f"Added '{book2.title}' to {user.username}'s favorites")
            else:
                print(f"'{book2.title}' is already in {user.username}'s favorites")
            
            # Commit the changes
            db.session.commit()
            print(f"{user.username} now has {len(user.favorite_books)} favorite books")
